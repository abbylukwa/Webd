   131→/* ---------- device preview with real iframe ---------- */
   132→function DevicePreview({ siteId, width }: { siteId: string; width: number }) {
   133→  const stageRef = useRef<HTMLDivElement>(null);
   134→  const [scale, setScale] = useState(1);
   135→  useLayoutEffect(() => {
   136→    const el = stageRef.current;
   137→    if (!el) return;
   138→    const update = () => setScale(Math.min(1, (el.clientWidth - 8) / width));
   139→    update();
   140→    const ro = new ResizeObserver(update);
   141→    ro.observe(el);
   142→    return () => ro.disconnect();
   143→  }, [width]);
   144→  const H = 600;
   145→  return (
   146→    <div ref={stageRef} className="w-full overflow-hidden">
   147→      <div className="mx-auto overflow-hidden rounded-xl border border-zinc-700 bg-white shadow-2xl" style={{ width: width * scale, height: H }}>
   148→        <iframe
   149→          src={`/api/beast/site/${siteId}`}
   150→          title="site preview"
   151→          style={{ width, height: H / scale, transform: `scale(${scale})`, transformOrigin: '0 0', border: 'none' }}
   152→        />
   153→      </div>
   154→    </div>
   155→  );
   156→}
