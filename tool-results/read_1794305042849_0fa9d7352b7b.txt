    51→function fmtEta(min: number | null): string {
    52→  if (min == null) return "—";
    53→  if (min < 1) return "moments";
    54→  if (min < 60) return `≈ ${Math.round(min)}m`;
    55→  const h = Math.floor(min / 60);
    56→  if (h < 36) return `≈ ${h}h ${Math.round(min % 60)}m`;
    57→  return `≈ ${Math.round(min / 1440)} days`;
    58→}

    61→function PhaseBar({ job, phases }: { job: JobView; phases: { key: string; label: string; minutes: number }[] }) {

    65→        const done = i < job.phaseIndex;
    66→        const current = i === job.phaseIndex;
    67→        return (
    68→          <div
    69→            key={p.key}
    70→            title={`${p.label} · ${p.minutes} beast-min${done ? " · done" : current ? " · in progress" : ""}`}

    76→  );

   518→// ─── Settings dialog — copyright, pace, API key vault ─────────────────
   519→function SettingsDialog({ open, onOpenChange, data, onSaved }: { open: boolean; onOpenChange: (o: boolean) => void; data: StudioState | null; onSaved: () => void }) {
   520→  const { toast } = useToast();
   521→  const [holder, setHolder] = useState("");
   522→  const [note, setNote] = useState("");
   523→  const [pace, setPace] = useState("6");
   524→  const [benches, setBenches] = useState("3");
   525→  const [respawn, setRespawn] = useState(true);
   526→  const [newKey, setNewKey] = useState("");
   527→  const [keyLabel, setKeyLabel] = useState("");
   528→  const [vault, setVault] = useState<VaultView[]>([]);
   529→  const [saving, setSaving] = useState(false);
   530→
   531→  useEffect(() => {
   532→    if (open && data) {
   533→      setHolder(data.settings.copyrightHolder);
   534→      setNote(data.settings.copyrightNote);
   535→      setPace(String(data.settings.pace));
   536→      setBenches(String(data.settings.workbenches));
   537→      setRespawn(data.settings.autoRespawn);
   538→      fetch("/api/settings").then((r) => r.json()).then((d) => setVault(d.vault ?? [])).catch(() => {});
   539→    }
   540→  }, [open, data]);
   541→
   542→  const save = async () => {
   543→    setSaving(true);
   544→    try {
   545→      const res = await fetch("/api/settings", {
   546→        method: "POST",
   547→        headers: { "content-type": "application/json" },
   548→        body: JSON.stringify({
   549→          copyrightHolder: holder,
   550→          copyrightNote: note,
   551→          pace: Number(pace),
   552→          workbenches: Number(benches),
   553→          autoRespawn: respawn,
   554→          newApiKey: newKey || undefined,
   555→          newKeyLabel: keyLabel || undefined,
   556→        }),
   557→      });
   558→      const d = await res.json();
   559→      if (!res.ok) throw new Error(d.error ?? "Save failed");
   560→      setVault(d.vault ?? []);
   561→      setNewKey("");
   562→      setKeyLabel("");
   563→      toast({ title: "Studio settings saved", description: newKey ? "Key stored server-side — only you can see the masked version." : "The Beast adjusts immediately." });
   564→      onSaved();
   565→    } catch (e) {
   566→      toast({ title: "Could not save", description: e instanceof Error ? e.message : "Try again.", variant: "destructive" });
   567→    } finally {
   568→      setSaving(false);
   569→    }
   570→  };
   571→
   572→  const removeKey = async (id: string) => {
   573→    await fetch("/api/settings", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ removeKeyId: id }) });
   574→    const d = await fetch("/api/settings").then((r) => r.json());
   575→    setVault(d.vault ?? []);
   576→    onSaved();
   577→  };
   578→
   579→  return (
   580→    <Dialog open={open} onOpenChange={onOpenChange}>
   581→      <DialogContent className="max-h-[85vh] max-w-lg overflow-y-auto border-stone-800 bg-stone-950 sm:max-w-lg">
   582→        <DialogHeader>
   583→          <DialogTitle className="font-[family-name:var(--font-fraunces)] text-stone-100">Studio settings</DialogTitle>
   584→          <DialogDescription className="text-stone-500">Copyright stamp, the Beast&apos;s pace, and the API key pool.</DialogDescription>
   585→        </DialogHeader>
   586→
   587→        <div className="grid gap-4 py-2">
   588→          <div className="grid gap-1.5">
   589→            <Label htmlFor="holder" className="text-stone-300">Copyright holder (stamped on every site)</Label>
   590→            <Input id="holder" value={holder} onChange={(e) => setHolder(e.target.value)} className="border-stone-700 bg-stone-900 text-stone-200" placeholder="Your name or company" />
   591→          </div>
   592→          <div className="grid gap-1.5">
   593→            <Label htmlFor="note" className="text-stone-300">Copyright note</Label>
   594→            <Input id="note" value={note} onChange={(e) => setNote(e.target.value)} className="border-stone-700 bg-stone-900 text-stone-200" placeholder="All rights reserved." />
   595→          </div>
   596→
   597→          <div className="grid grid-cols-2 gap-3">
   598→            <div className="grid gap-1.5">
   599→              <Label className="text-stone-300">Pace (beast-min / real-min)</Label>
   600→              <Select value={pace} onValueChange={setPace}>
   601→                <SelectTrigger className="border-stone-700 bg-stone-900 text-stone-200"><SelectValue /></SelectTrigger>
   602→                <SelectContent className="border-stone-800 bg-stone-900 text-stone-200">
   603→                  <SelectItem value="1">1 — contemplative (site ≈ 15h)</SelectItem>
   604→                  <SelectItem value="6">6 — steady (site ≈ 2.5h)</SelectItem>
   605→                  <SelectItem value="24">24 — driven (site ≈ 38m)</SelectItem>
   606→                  <SelectItem value="120">120 — obsessed (site ≈ 8m)</SelectItem>
   607→                </SelectContent>
   608→              </Select>
   609→            </div>
   610→            <div className="grid gap-1.5">
   611→              <Label className="text-stone-300">Workbenches</Label>
   612→              <Select value={benches} onValueChange={setBenches}>
   613→                <SelectTrigger className="border-stone-700 bg-stone-900 text-stone-200"><SelectValue /></SelectTrigger>
   614→                <SelectContent className="border-stone-800 bg-stone-900 text-stone-200">
   615→                  {[1, 2, 3, 4, 5, 6].map((n) => <SelectItem key={n} value={String(n)}>{n} site{n > 1 ? "s" : ""} at once</SelectItem>)}
   616→                </SelectContent>
   617→              </Select>
   618→            </div>
   619→          </div>
   620→          <div className="flex items-center justify-between rounded-xl border border-stone-800 bg-stone-900/50 p-3">
   621→            <div>
   622→              <p className="text-sm text-stone-300">Auto-respawn commissions</p>
   623→              <p className="text-xs text-stone-500">Keep every bench full, forever. Untick to stop taking new work.</p>
   624→            </div>
   625→            <Switch checked={respawn} onCheckedChange={setRespawn} className="data-[state=checked]:bg-amber-400" />
   626→          </div>
   627→
   628→          <Separator className="bg-stone-800" />
   629→
   630→          <div>
   631→            <div className="mb-1 flex items-center gap-2">
   632→              <KeyRound className="h-4 w-4 text-amber-400" />
   633→              <Label className="text-stone-300">AI key pool (optional)</Label>

   635→            <p className="mb-2 text-xs leading-relaxed text-stone-500">
   636→              Same doctrine as your whatsapp-qr-app: numbered slots, provider auto-detected from the key&apos;s shape, round-robin across healthy keys, cooldown on failure, graceful fallback. Keys are stored server-side and never returned raw.
   637→            </p>
   638→            <div className="flex gap-2">
   639→              <Input value={keyLabel} onChange={(e) => setKeyLabel(e.target.value)} placeholder="Label (e.g. OpenRouter)" className="w-36 border-stone-700 bg-stone-900 text-stone-200" />
   640→              <Input value={newKey} onChange={(e) => setNewKey(e.target.value)} placeholder="Paste key (sk-…, gsk_…, AIza…)" type="password" className="flex-1 border-stone-700 bg-stone-900 text-stone-200" />
   641→              <Button size="icon" variant="outline" onClick={save} disabled={!newKey || saving} className="shrink-0 border-stone-700 bg-stone-900 hover:bg-stone-800 hover:text-amber-300" aria-label="Add key">
   642→                <Plus className="h-4 w-4" />
   643→              </Button>
   644→            </div>
   645→            <div className="mt-3 grid gap-2">
   646→              {vault.length === 0 && <p className="text-xs text-stone-600">Vault empty — the Beast writes its own copy (and says so in the log when it does).</p>}

   649→                  <span className="rounded bg-stone-800 px-1.5 py-0.5 font-mono text-[10px] text-amber-300">{k.provider}</span>

   653→                  <button onClick={() => removeKey(k.id)} aria-label={`Remove ${k.label}`} className="text-stone-600 transition-colors hover:text-rose-400"><X className="h-3.5 w-3.5" /></button>
   654→                </div>
   655→              ))}
   656→            </div>
   657→          </div>
   658→        </div>
   659→
   660→        <DialogFooter>
   661→          <Button onClick={save} disabled={saving} className="bg-amber-400 font-semibold text-stone-950 hover:bg-amber-300">
   662→            {saving ? "Saving…" : "Save settings"}
   663→          </Button>

   665→      </DialogContent>
   666→    </Dialog>
