   173→   173→function pickN<T>(seed: string, salt: number, arr: T[]): T {
   174→   174→  let h = salt;
   175→   175→  for (const c of seed) h = (h * 31 + c.charCodeAt(0)) >>> 0;
   176→   176→  return arr[h % arr.length];
   177→   177→}

   181→   181→function solver(kind: TaskKind, site: V3Site, ctx: TaskCtx): { output?: unknown; note: string; precision: number } {
   182→   182→  const probe = designFromSeed(site.seed);
   183→   183→  switch (kind) {
   184→   184→    case 'brief.compose': {
   185→   185→      const tone = pickN(site.seed, 1, TONES);
   186→   186→      return {
   187→   187→        output: {
   188→   188→          audience: `${probe.niche.label} regulars who notice details`,
   189→   189→          tone,
   190→   190→          promise: `${probe.niche.heroLines[0]} ${probe.brand} keeps that promise in ${probe.city}.`,
   191→   191→          cityStory: `${probe.niche.about[0]} — and we do it in ${probe.city}.`,
   192→   192→        },
   193→   193→        note: 'deterministic solver — brief composed from niche grammar', precision: 0.9,
   194→   194→      };
   195→   195→    }
   196→   196→    case 'palette.lock':
   197→   197→      return { output: { name: site.palette.name, harmony: site.palette.harmony }, note: 'deterministic solver — harmony math + WCAG battery already passed', precision: 1 };
   198→   198→    case 'layout.select': {
   199→   199→      const device = (site.tasks.find((t) => t.kind === 'layout.select' && t.id === site.tasks.find((x) => x.status === 'running')?.id)?.device ?? 'D3') as DeviceId;
   200→   200→      const pool = layoutsForDevice(device);
   201→   201→      const lay = pickN(site.seed + device, 7, pool);
   202→   202→      return {
   203→   203→        output: { id: lay.id, why: `${lay.name.split(' — ')[0]} — ${lay.arrangement} (source: ${lay.source})` },
   204→   204→        note: `deterministic solver — brief-fit pick from ${pool.length} ${device} layouts`, precision: 0.9,
   205→   205→      };
   206→   206→    }
   207→   207→    case 'page.plan': {
   208→   208→      // the values defaults are born BEFORE the map is drawn, so the catalog
   209→   209→      // size can earn the Shop page deterministically
   210→   210→      if (!site.art.values) {
   211→   211→        const probe = designFromSeed(site.seed);
   212→   212→        const rng = seedFrom(site.seed + '-values');
   213→   213→        site.art.values = defaultValues(site.seed, site.brand, probe.niche.label, probe.city, probe.niche.place, probe.niche.menuTitle, probe.niche.priceBase, probe.niche.services, rng);
   214→   214→      }
   215→   215→      const pages = planPagesFor(site);
   216→   216→      return { output: { pages }, note: `deterministic solver — ${pages.length} pages planned (multi-page mandate)`, precision: 1 };
   217→   217→    }
   218→   218→    case 'heading.compose': {
   219→   219→      const t = site.tasks.find((x) => x.status === 'running');
   220→   220→      const pageKey = t?.pageKey ?? 'about';
   221→   221→      const slot = t?.slot ?? 0;
   222→   222→      const text = headingFor(site, pageKey, slot);
   223→   223→      return { output: { text, alt: text }, note: 'deterministic solver — heading from niche grammar, uniqueness-checked', precision: 0.9 };
   224→   224→    }
   225→   225→    case 'section.copy': {
   226→   226→      const body = `${probe.niche.about[1] ?? probe.niche.about[0]}. ${pickN(site.seed, 3, ['Every piece is checked before it leaves the bench.', 'We keep two slots open each week for first-time visitors.', 'Ask us anything before you commit — the answer is free.'])} In ${probe.city} since ${2016 + (site.seed.charCodeAt(0) % 8)}.`;
   227→   227→      return { output: { kicker: pickN(site.seed, 4, ['The work', 'How it goes', 'Why here']), body }, note: `deterministic solver — ${body.split(/\s+/).length} words, one fact`, precision: 0.85 };
   228→   228→    }
   229→   229→    case 'image.select':
   230→   230→      return { output: { deferredSlots: true }, note: 'image resolution runs in the builder (network task)', precision: 0.5 };
   231→   231→    case 'catalog.build': {
   232→   232→      const vals = site.art.values ?? ({} as SiteValues);
   233→   233→      const catalog = (vals.products ?? []).slice(0, 30);
   234→   234→      return { output: { catalog }, note: `deterministic solver — ${catalog.length} products, caps respected`, precision: 1 };
   235→   235→    }
   236→   236→    case 'contact.channels': {
   237→   237→      const vals = site.art.values ?? ({} as SiteValues);
   238→   238→      const phones = (vals.phones ?? []).length;
   239→   239→      const socials = (vals.socials ?? []).length;
   240→   240→      return { output: { phones, socials }, note: `deterministic solver — ${phones} labelled numbers, ${socials} social platforms, JSON-LD contactPoints wired`, precision: 1 };
   241→   241→    }
   242→   242→    case 'values.freeze': {
   243→   243→      const vals = site.art.values ?? ({} as SiteValues);
   244→   244→      const res = validateValues(vals);
   245→   245→      return { output: res.clean, note: res.ok ? 'values validated — caps respected' : `issues: ${res.issues.map((i) => i.field).join(',')}`, precision: res.ok ? 1 : 0.6 };
   246→   246→    }
   247→   247→    default:
   248→   248→      return { note: 'deterministic solver — accepted', precision: 0.9 };
   249→   249→  }
   250→   250→}

   308→   308→export async function runNextTask(ctx: TaskCtx, site: V3Site): Promise<string> {
   309→   309→  const task = site.tasks.find((t) => t.status === 'queued');
   310→   310→  if (!task) return `${site.brand}: no queued task (ledger complete)`;
   311→   311→  task.status = 'running';
   312→   312→  const { designFromSeed: dfs } = require('./render') as typeof import('./render');
   313→   313→  const probe = dfs(site.seed);
   314→   314→
   315→   315→  let applied = '';
   316→   316→  let precision = 0;
   317→   317→  let mode: 'ai' | 'solver' = 'solver';
   318→   318→  let provider = 'deterministic-solver';
   319→   319→  let verdict: 'ACCEPT' | 'REFINE' | 'DEFER' = 'ACCEPT';
   320→   320→  let interpNote = '';
   321→   321→
   322→   322→  // AI path for text-bearing kinds when the vault has usable keys
   323→   323→  const aiKinds: TaskKind[] = ['brief.compose', 'heading.compose', 'section.copy'];
   324→   324→  if (aiEnabled() && aiKinds.includes(task.kind)) {
   325→   325→    const stateFacts = [
   326→   326→      `brand: ${site.brand} — ${site.nicheLabel} in ${site.city}`,
   327→   327→      `brief: ${site.art.brief ? JSON.stringify(site.art.brief) : 'not yet locked'}`,
   328→   328→      `frozen headings so far: ${JSON.stringify(site.art.headings ?? {}).slice(0, 500)}`,
   329→   329→      `niche vocabulary: ${probe.niche.services.join(', ')}`,
   330→   330→    ];
   331→   331→    const contract =
   332→   332→      task.kind === 'heading.compose' ? 'JSON only: {"text": "<=60 chars heading", "alt": "accessibility alt"}'
   333→   333→      : task.kind === 'section.copy' ? 'JSON only: {"kicker": "<=40 chars", "body": "25-70 words"}'
   334→   334→      : 'JSON only: {"audience": "...", "tone": "...", "promise": "...", "cityStory": "..."}';
   335→   335→    const env = buildEnvelope({
   336→   336→      role: `You are the ${task.kind.startsWith('heading') ? 'headline editor' : task.kind.startsWith('brief') ? 'strategy director' : 'senior copywriter'} of a slow, precise design studio.`,
   337→   337→      why: task.why,
   338→   338→      rules: task.rules,
   339→   339→      knowledge: task.knowledge,
   340→   340→      state: stateFacts,
   341→   341→      contract,
   342→   342→    });
   343→   343→    const res = await askAi(env, task.kind);
   344→   344→    if (res) {
   345→   345→      mode = 'ai';
   346→   346→      provider = res.turn.provider;
   347→   347→      verdict = res.turn.verdict;
   348→   348→      interpNote = res.turn.interpretation;
   349→   349→      if (verdict === 'ACCEPT' && res.artifact) {
   350→   350→        applyOutput(ctx, site, task, res.artifact);
   351→   351→        applied = interpNote;
   352→   352→        precision = 1;
   353→   353→      } else if (verdict === 'REFINE') {
   354→   354→        task.attempts += 1;
   355→   355→        task.note = interpNote;
   356→   356→        precision = 0.35;
   357→   357→        if (task.attempts >= 3) { verdict = 'DEFER'; }
   358→   358→      }
   359→   359→      ctx.logTurn({
   360→   360→        id: `${task.id}-${Date.now()}`, taskId: task.id, taskTitle: task.title, provider, mode,
   361→   361→        promptTitle: task.why, envelope: env.prompt.slice(0, 1500), raw: res.turn.raw,
   362→   362→        interpretation: interpNote, verdict,
   363→   363→        action: verdict === 'ACCEPT' ? 'output frozen into artifacts' : verdict === 'REFINE' ? `re-queued with critique (attempt ${task.attempts}/3)` : 'deferred to deterministic solver',
   364→   364→        precision, at: new Date().toISOString(),
   365→   365→      });
   366→   366→    }
   367→   367→  }
   368→   368→
   369→   369→  if (mode === 'solver') {
   370→   370→    // deterministic floor (also the DEFER landing)
   371→   371→    const out = solver(task.kind, site, ctx);
   372→   372→    applyOutput(ctx, site, task, out.output);
   373→   373→    precision = out.precision;
   374→   374→    interpNote = out.note;
   375→   375→    provider = 'deterministic-solver';
   376→   376→    verdict = 'ACCEPT';
   377→   377→    ctx.logTurn({
   378→   378→      id: `${task.id}-${Date.now()}`, taskId: task.id, taskTitle: task.title, provider, mode: 'solver',
   379→   379→      promptTitle: task.why, envelope: task.rules.join(' | ').slice(0, 800), raw: JSON.stringify(out.output ?? {}).slice(0, 500),
   380→   380→      interpretation: out.note, verdict: 'ACCEPT', action: 'output frozen into artifacts', precision: out.precision, at: new Date().toISOString(),
   381→   381→    });
   382→   382→  }
   383→   383→
   384→   384→  if (verdict === 'REFINE') {
   385→   385→    task.status = 'queued';
   386→   386→    return `${site.brand}: ${task.title} — refine (attempt ${task.attempts}/3): ${task.note ?? 'critique'}`;
   387→   387→  }
   388→   388→  if (verdict === 'ACCEPT') {
   389→   389→    task.status = 'accepted';
   390→   390→    task.precision = precision;
   391→   391→    task.acceptedDay = ctx.day;
   392→   392→    // page.plan decomposes into the per-page sub-task tree
   393→   393→    if (task.kind === 'page.plan') {
   394→   394→      site.tasks.push(...expandAfterPlan(site).map((t) => ({ ...t, siteId: site.id })));
   395→   395→    }
   396→   396→    applied = applied || interpNote;
   397→   397→    return `${site.brand}: ${task.title} — ${applied}`;
   398→   398→  }
   399→   399→  if (verdict === 'DEFER') {
   400→   400→    task.status = 'accepted'; // solver already produced the artifact above
   401→   401→    task.precision = Math.max(task.precision, 0.5);
   402→   402→    return `${site.brand}: ${task.title} — AI unavailable after ${task.attempts} attempts, deterministic solver finished it`;
   403→   403→  }
   404→   404→  return `${site.brand}: ${task.title} — refine (${task.note ?? 'critique'})`;
   405→   405→}

   407→   407→function applyOutput(ctx: TaskCtx, site: V3Site, task: Task, output: unknown): void {
   408→   408→  site.art = site.art ?? {};
   409→   409→  switch (task.kind) {
   410→   410→    case 'brief.compose':
   411→   411→      site.art.brief = output as SiteArt['brief'];
   412→   412→      break;
   413→   413→    case 'layout.select':
   414→   414→      site.art.layouts = site.art.layouts ?? {};
   415→   415→      site.art.layouts[task.device ?? 'D3'] = output as { id: string; why: string };
   416→   416→      break;
   417→   417→    case 'page.plan': {
   418→   418→      const pages = (output as { pages: PagePlanEntry[] }).pages;
   419→   419→      site.art.pages = pages;
   420→   420→      // values defaults are born with the plan (they drive catalog + channels)
   421→   421→      if (!site.art.values) {
   422→   422→        const probe = designFromSeed(site.seed);
   423→   423→        const rng = seedFrom(site.seed + '-values');
   424→   424→        site.art.values = defaultValues(site.seed, site.brand, probe.niche.label, probe.city, probe.niche.place, probe.niche.menuTitle, probe.niche.priceBase, probe.niche.services, rng);
   425→   425→      }
   426→   426→      break;
   427→   427→    }
   428→   428→    case 'heading.compose': {
   429→   429→      site.art.headings = site.art.headings ?? {};
   430→   430→      const arr = site.art.headings[task.pageKey ?? 'about'] ?? [];
   431→   431→      arr[task.slot ?? 0] = (output as { text: string }).text;
   432→   432→      site.art.headings[task.pageKey ?? 'about'] = arr;
   433→   433→      break;
   434→   434→    }
   435→   435→    case 'section.copy': {
   436→   436→      site.art.copy = site.art.copy ?? {};
   437→   437→      const arr = site.art.copy[task.pageKey ?? 'about'] ?? [];
   438→   438→      arr.push(output as { kicker: string; body: string });
   439→   439→      site.art.copy[task.pageKey ?? 'about'] = arr;
   440→   440→      break;
   441→   441→    }
   442→   442→    case 'catalog.build':
   443→   443→      site.art.catalog = (output as { catalog: Product[] }).catalog;
   444→   444→      break;
   445→   445→    case 'contact.channels':
   446→   446→      site.art.channelsAccepted = true;
   447→   447→      break;
   448→   448→    case 'values.freeze':
   449→   449→      site.art.values = output as SiteValues;
   450→   450→      break;
   451→   451→    case 'page.build':
   452→   452→    case 'page.review':
   453→   453→    case 'qa.audit':
   454→   454→    case 'site.stamp':
   455→   455→    case 'github.publish':
   456→   456→    case 'image.select':
   457→   457→    case 'palette.lock':
   458→   458→      // handled by the bespoke builders below
   459→   459→      break;
   460→   460→    default:
   461→   461→      break;
   462→   462→  }
   463→   463→  void fs; void path;
   464→   464→}

   467→   467→export async function executeIoTask(ctx: TaskCtx, site: V3Site, task: Task): Promise<string | null> {
   468→   468→  const dir = ctx.siteDir(site.id);
   469→   469→  if (task.kind === 'image.select') {
   470→   470→    fs.mkdirSync(path.join(dir, 'assets'), { recursive: true });
   471→   471→    const pending = imageSlotsFor(site).filter((s) => !(site.art.images ?? []).some((r) => r.slotId === s.slotId && r.status !== 'gradient'));
   472→   472→    const one = pending[0];
   473→   473→    if (!one) return null; // nothing pending — accept as complete
   474→   474→    const { records, newFetches } = await resolveImages(site.seed, siteNicheKey(site), site.nicheLabel, [one], site.art.images ?? [], (slotId, bytes) => {
   475→   475→      const file = `assets/${slotId}.jpg`;
   476→   476→      fs.writeFileSync(path.join(dir, file), bytes);
   477→   477→      return file;
   478→   478→    });
   479→   479→    site.art.images = [...(site.art.images ?? []).filter((r) => r.slotId !== one.slotId), ...records];
   480→   480→    const rec = records[0];
   481→   481→    return `${site.brand}: image ${one.slotId} — ${rec.status === 'live' ? `live CC image from ${rec.source} (${rec.license})` : rec.status === 'picsum' ? 'picsum fallback' : 'gradient fallback'} (+${newFetches} fetch)`;
   482→   482→  }
   483→   483→  if (task.kind === 'page.build' || task.kind === 'page.review') {
   484→   484→    const t = task;
   485→   485→    const pageKey = t.pageKey ?? 'home';
   486→   486→    const pages = site.art.pages ?? [];
   487→   487→    const art: RenderArt = {
   488→   488→      brief: site.art.brief, headings: site.art.headings, copy: site.art.copy,
   489→   489→      images: site.art.images, catalog: site.art.catalog ?? undefined, values: site.art.values ?? undefined, pages,
   490→   490→    };
   491→   491→    fs.mkdirSync(dir, { recursive: true });
   492→   492→    let file = 'index.html';
   493→   493→    let html = '';
   494→   494→    if (pageKey === 'home') {
   495→   495→      html = renderSite(site.seed, ctx.owner, { art }).html;
   496→   496→      // inject the page strip linking the inner pages (deterministic)
   497→   497→      const strip = `<div class="wrap" style="display:flex;flex-wrap:wrap;gap:10px;padding:14px 16px;border-bottom:1px solid var(--border)">${pages.filter((p) => p.key !== 'home').map((p) => `<a href="${p.file}" style="font-size:.85rem;font-weight:600;color:var(--accent)">${p.title} ↗</a>`).join('')}</div>`;
   498→   498→      html = html.replace('<main id="main">', `<main id="main">${strip}`);
   499→   499→      file = 'index.html';
   500→   500→    } else {
   501→   501→      file = pages.find((p) => p.key === pageKey)?.file ?? `${pageKey}.html`;
   502→   502→      html = renderInnerPage(site.seed, ctx.owner, pageKey, { art, currentKey: pageKey });
   503→   503→    }
   504→   504→    fs.writeFileSync(path.join(dir, file), html);
   505→   505→    const lines = html.split('\n').length;
   506→   506→    site.art.revisions = site.art.revisions ?? {};
   507→   507→    const revs = site.art.revisions[pageKey] ?? [];
   508→   508→    const prev = revs[revs.length - 1];
   509→   509→    const diff = prev ? countDiff(prev.v === 1 ? prev.lines : prev.lines, lines) : 0;
   510→   510→    revs.push({ v: revs.length + 1, lines, diff, at: new Date().toISOString() });
   511→   511→    site.art.revisions[pageKey] = revs.slice(-2);
   512→   512→    if (task.kind === 'page.build') return `${site.brand}: built ${file} (${lines} lines, D1–D5 media blocks, nav wired)`;
   513→   513→    return `${site.brand}: reviewed ${file} — rev${revs.length}, ${diff} changed lines, structure intact`;
   514→   514→  }
   515→   515→  if (task.kind === 'qa.audit') {
   516→   516→    const pages = ['index.html', ...(site.art.pages ?? []).filter((p) => p.key !== 'home').map((p) => p.file)];
   517→   517→    let passing = 0;
   518→   518→    for (const f of pages) {
   519→   519→      const p = path.join(dir, f);
   520→   520→      if (fs.existsSync(p)) {
   521→   521→        const html = fs.readFileSync(p, 'utf8');
   522→   522→        if (html.includes('@media (min-width:641px)') && html.includes('@media (min-width:1025px)') && html.includes('@media (min-width:1921px)')) passing += 1;
   523→   523→      }
   524→   524→    }
   525→   525→    const r = renderSite(site.seed, ctx.owner);
   526→   526→    site.qa = {
   527→   527→      contrast: r.palette.report.map((x) => ({ pair: x.pair, ratio: x.ratio, pass: x.pass })),
   528→   528→      breakpoints: ['@media (min-width:641px)', '@media (min-width:1025px)', '@media (min-width:1441px)', '@media (min-width:1921px)'],
   529→   529→      crap: [`Contrast — every pair WCAG-corrected`, `Repetition — one token set across ${pages.length} pages`, 'Alignment — container grid everywhere', 'Proximity — related blocks grouped'],
   530→   530→    };
   531→   531→    return `${site.brand}: QA audit — ${passing}/${pages.length} pages carry the full device matrix; contrast battery ${r.palette.report.every((x) => x.pass) ? 'clean' : 'flagged'}`;
   532→   532→  }
   533→   533→  if (task.kind === 'site.stamp') {
   534→   534→    const year = new Date().getFullYear();
   535→   535→    const base = site.nicheLabel.includes('architecture') || site.nicheLabel.includes('agency') ? 140 : site.nicheLabel.includes('tattoo') ? 90 : 60;
   536→   536→    site.price = sitePriceFor(site.seed, base);
   537→   537→    site.status = 'for-sale';
   538→   538→    site.copyright = `© ${year} ${ctx.owner}. All rights reserved.`;
   539→   539→    // re-render every page with the final owner stamp
   540→   540→    await rebuildAllPages(ctx, site);
   541→   541→    // write attribution + values into the folder
   542→   542→    fs.writeFileSync(path.join(dir, 'ATTRIBUTION.md'), attributionMarkdown(site.brand, site.art.images ?? []));
   543→   543→    fs.writeFileSync(path.join(dir, 'values.json'), JSON.stringify(site.art.values ?? {}, null, 1));
   544→   544→    return `${site.brand}: © stamped for ${ctx.owner} — on the shelf at $${site.price} (${(site.art.pages ?? []).length + 1} pages)`;
   545→   545→  }
   546→   546→  if (task.kind === 'github.publish') {
   547→   547→    const files = listSiteFiles(ctx.siteDir(site.id)).map((f) => ({
   548→   548→      path: f.rel,
   549→   549→      content: f.binary ? f.bytes!.toString('base64') : f.text!,
   550→   550→      encoding: f.binary ? ('base64' as const) : ('utf8' as const),
   551→   551→    }));
   552→   552→    const res = await publishSiteFiles(site.brand, site.seed, files, `${site.brand} — ${site.nicheLabel} in ${site.city}`);
   553→   553→    if (res.ok && res.repo) {
   554→   554→      site.art.git = { repo: res.repo, url: res.url!, pushedAt: new Date().toISOString() };
   555→   555→      return `${site.brand}: published to github.com/${res.repo} (one commit, ${files.length} files)`;
   556→   556→    }
   557→   557→    task.status = 'deferred';
   558→   558→    task.note = res.reason;
   559→   559→    return `${site.brand}: GitHub publish deferred — ${res.reason}`;
   560→   560→  }
   561→   561→  return null;
   562→   562→}

   588→   588→export function listSiteFiles(dir: string): { rel: string; text?: string; bytes?: Buffer; binary?: boolean }[] {
   589→   589→  const out: { rel: string; text?: string; bytes?: Buffer; binary?: boolean }[] = [];
   590→   590→  const walk = (d: string, prefix: string) => {
   591→   591→    if (!fs.existsSync(d)) return;
   592→   592→    for (const f of fs.readdirSync(d, { withFileTypes: true })) {
   593→   593→      const full = path.join(d, f.name);
   594→   594→      const rel = prefix ? `${prefix}/${f.name}` : f.name;
   595→   595→      if (f.isDirectory()) walk(full, rel);
   596→   596→      else {
   597→   597→        const bytes = fs.readFileSync(full);
   598→   598→        const binary = /\.(jpg|jpeg|png|gif|webp|ico)$/i.test(f.name);
   599→   599→        out.push({ rel, bytes, binary, text: binary ? undefined : bytes.toString('utf8') });
   600→   600→      }
   601→   601→    }
   602→   602→  };
   603→   603→  walk(dir, '');
   604→   604→  return out;
   605→   605→}

   607→   607→function sitePriceFor(seed: string, nicheKeyPriceBase: number): number {
   608→   608→  const probe = designFromSeed(seed);
   609→   609→  const complexity = 1 + (probe.design.heroVariant === 'cinematic' ? 0.25 : 0) + (probe.design.motif === 'blobs' ? 0.1 : 0) + (probe.design.spaceRhythm === 'generous' ? 0.1 : 0);
   610→   610→  const rng = seedFrom(seed + '-price');
   611→   611→  return Math.round((nicheKeyPriceBase * 9 * complexity + rng.int(40, 160)) / 10) * 10;
   612→   612→}
