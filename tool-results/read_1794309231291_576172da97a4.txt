    27→/* ---------- design spec from seed ---------- */
    28→export function designFromSeed(seed: string): { rng: RNG; niche: Niche; brand: string; city: string; design: DesignSpec; palette: Palette } {
    29→  const rng = seedFrom(seed);
    30→  const niche = rng.pick(NICHES);
    31→  const city = rng.pick(CITIES);
    32→
    33→  // naming grammar
    34→  const pattern = rng.pick(['theAdjNoun', 'compound', 'surname', 'plainNoun'] as const);
    35→  const adj = rng.pick(ADJECTIVES);
    36→  const noun = rng.pick(niche.nouns);
    37→  const brand =
    38→    pattern === 'theAdjNoun' ? `The ${adj} ${noun}`
    39→    : pattern === 'compound' ? `${adj}${noun}`
    40→    : pattern === 'surname' ? `${rng.pick(SURNAMES)}'s ${niche.label.split(' ')[0].replace(/^\w/, (c) => c.toUpperCase())}`
    41→    : `${noun} & ${rng.pick(ADJECTIVES)}`;
    42→
    43→  const palette = buildPalette(rng);
    44→  const design: DesignSpec = {
    45→    heroVariant: rng.pick(['poster', 'split', 'cinematic', 'editorial', 'diagonal', 'marquee'] as HeroVariant[]),
    46→    navVariant: rng.pick(['drawer', 'inline', 'pill', 'topline'] as NavVariant[]),
    47→    motif: rng.pick(['grain', 'blobs', 'grid', 'rings', 'waves', 'dots', 'arc'] as Motif[]),
    48→    radius: rng.pick([0, 2, 4, 8, 12, 16, 20]),
    49→    btnRadius: rng.pick([0, 4, 8, 999]),
    50→    spaceRhythm: rng.pick(['compact', 'cozy', 'generous'] as const),
    51→    sectionOrder: [],
    52→    eyebrow: rng.pick(['dot', 'line', 'bracket', 'number'] as const),
    53→    displayCase: rng.pick(['upper', 'title', 'lower'] as const),
    54→    fontHead: (function(){ const fh = rng.pick(HEAD_FONTS); return fh; })(),
    55→    fontBody: rng.pick(BODY_FONTS),
    56→    reveal: rng.chance(0.8),
    57→    darkNav: palette.theme === 'dark' ? rng.chance(0.5) : rng.chance(0.15),
    58→  };
    59→  return { rng, niche, brand, city, design, palette };
    60→}

    63→function motifCss(motif: Motif, p: Palette): string {
    64→  switch (motif) {
    65→    case 'grain': return `.ambient{opacity:.5;background-image:radial-gradient(${p.border} .8px,transparent .8px),radial-gradient(${p.border} .8px,transparent .8px);background-size:22px 22px,22px 22px;background-position:0 0,11px 11px;}`;
    66→    case 'blobs': return `.ambient{opacity:.6;background:radial-gradient(600px 400px at 12% 8%,${p.accent}22,transparent 60%),radial-gradient(700px 500px at 88% 20%,${p.accent2}1e,transparent 65%),radial-gradient(500px 600px at 50% 100%,${p.accent}14,transparent 60%);}`;
    67→    case 'grid': return `.ambient{opacity:.45;background-image:linear-gradient(${p.border} 1px,transparent 1px),linear-gradient(90deg,${p.border} 1px,transparent 1px);background-size:56px 56px;}`;
    68→    case 'rings': return `.ambient{opacity:.5;background:repeating-radial-gradient(circle at 85% 10%,transparent 0 46px,${p.border} 46px 47px);}`;
    69→    case 'waves': return `.ambient{opacity:.5;background:repeating-linear-gradient(115deg,transparent 0 38px,${p.border} 38px 39px);}`;
    70→    case 'dots': return `.ambient{opacity:.55;background-image:radial-gradient(${p.inkSoft}55 1.2px,transparent 1.2px);background-size:26px 26px;}`;
    71→    default: return `.ambient{opacity:.5;background:radial-gradient(120% 90% at 50% -20%,transparent 60%,${p.accent}18 100%);}`;
    72→  }
    73→}

   274→/* ---------- render ---------- */
   275→export function renderSite(seed: string, owner: string): RenderedSite {
   276→  const { rng, niche, brand, city, design, palette } = designFromSeed(seed);
   277→  const year = new Date().getFullYear();
   278→  const email = `hello@${brandSlug(brand)}`;
   279→  const phone = `+1 (${rng.int(200, 989)}) ${rng.int(200, 989)}-${rng.int(1000, 9999)}`;
   280→
   281→  // section order: hero first; pick 6 middles; cta before footer
   282→  const pool = rng.shuffle(['about', 'services', 'gallery', 'menu', 'testimonials', 'stats', 'faq', 'contact']);
   283→  const middles = pool.slice(0, 6);
   284→  if (!middles.includes('contact')) middles[middles.length - 1] = 'contact';
   285→  design.sectionOrder = ['hero', ...middles.slice(0, 5), 'cta'];
   286→
   287→  const navAnchors = design.sectionOrder
   288→    .filter((s) => !['hero', 'cta'].includes(s))
   289→    .map((s) => ({ id: s, label: s === 'services' ? 'Services' : s === 'gallery' ? 'Work' : s === 'menu' ? niche.menuTitle.split(' ')[0] : s === 'testimonials' ? 'Reviews' : s === 'stats' ? 'Numbers' : s === 'faq' ? 'FAQ' : s === 'about' ? 'About' : 'Visit' }));
