   129→/* ---------- device preview with real iframe ---------- */
   130→function DevicePreview({ siteId, width }: { siteId: string; width: number }) {
   131→  const stageRef = useRef<HTMLDivElement>(null);
   132→  const [scale, setScale] = useState(1);
   133→  useLayoutEffect(() => {
   134→    const el = stageRef.current;
   135→    if (!el) return;
   136→    const update = () => setScale(Math.min(1, (el.clientWidth - 8) / width));
   137→    update();
   138→    const ro = new ResizeObserver(update);
   139→    ro.observe(el);
   140→    return () => ro.disconnect();
   141→  }, [width]);
   142→  const H = 600;
   143→  return (
   144→    <div ref={stageRef} className="w-full overflow-hidden">
   145→      <div className="mx-auto overflow-hidden rounded-xl border border-zinc-700 bg-white shadow-2xl" style={{ width: width * scale, height: H }}>
   146→        <iframe
   147→          src={`/api/beast/site/${siteId}`}
   148→          title="site preview"
   149→          style={{ width, height: H / scale, transform: `scale(${scale})`, transformOrigin: '0 0', border: 'none' }}
   150→        />
   151→      </div>
   152→    </div>
   153→  );
   154→}
