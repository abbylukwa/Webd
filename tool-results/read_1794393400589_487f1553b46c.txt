    28→/* ---------- design spec from seed ---------- */
    29→export function designFromSeed(seed: string): { rng: RNG; niche: Niche; brand: string; city: string; design: DesignSpec; palette: Palette } {
    30→  const rng = seedFrom(seed);
    31→  const niche = rng.pick(NICHES);
    32→  const city = rng.pick(CITIES);
    33→
    34→  // naming grammar
    35→  const pattern = rng.pick(['theAdjNoun', 'compound', 'surname', 'plainNoun'] as const);
    36→  const adj = rng.pick(ADJECTIVES);
    37→  const noun = rng.pick(niche.nouns);
    38→  const brand =
    39→    pattern === 'theAdjNoun' ? `The ${adj} ${noun}`
    40→    : pattern === 'compound' ? `${adj}${noun}`
    41→    : pattern === 'surname' ? `${rng.pick(SURNAMES)}'s ${niche.label.split(' ')[0].replace(/^\w/, (c) => c.toUpperCase())}`
    42→    : `${noun} & ${rng.pick(ADJECTIVES)}`;
    43→
    44→  const palette = buildPalette(rng);
    45→  const design: DesignSpec = {
    46→    heroVariant: rng.pick(['poster', 'split', 'cinematic', 'editorial', 'diagonal', 'marquee'] as HeroVariant[]),
    47→    navVariant: rng.pick(['drawer', 'inline', 'pill', 'topline'] as NavVariant[]),
    48→    motif: rng.pick(['grain', 'blobs', 'grid', 'rings', 'waves', 'dots', 'arc'] as Motif[]),
    49→    radius: rng.pick([0, 2, 4, 8, 12, 16, 20]),
    50→    btnRadius: rng.pick([0, 4, 8, 999]),
    51→    spaceRhythm: rng.pick(['compact', 'cozy', 'generous'] as const),
    52→    sectionOrder: [],
    53→    eyebrow: rng.pick(['dot', 'line', 'bracket', 'number'] as const),
    54→    displayCase: rng.pick(['upper', 'title', 'lower'] as const),
    55→    fontHead: (function(){ const fh = rng.pick(HEAD_FONTS); return fh; })(),
    56→    fontBody: rng.pick(BODY_FONTS),
    57→    reveal: rng.chance(0.8),
    58→    darkNav: palette.theme === 'dark' ? rng.chance(0.5) : rng.chance(0.15),
    59→  };
    60→  return { rng, niche, brand, city, design, palette };
    61→}

    64→// Motifs are painted with token vars + color-mix (never hardcoded hex), so the
    65→// whole surface re-tints automatically under prefers-color-scheme: dark.
    66→function motifCss(motif: Motif): string {
    67→  switch (motif) {
    68→    case 'grain': return `.ambient{opacity:.5;background-image:radial-gradient(var(--border) .8px,transparent .8px),radial-gradient(var(--border) .8px,transparent .8px);background-size:22px 22px,22px 22px;background-position:0 0,11px 11px;}`;
    69→    case 'blobs': return `.ambient{opacity:.6;background:radial-gradient(600px 400px at 12% 8%,color-mix(in srgb,var(--accent) 13%,transparent),transparent 60%),radial-gradient(700px 500px at 88% 20%,color-mix(in srgb,var(--accent2) 12%,transparent),transparent 65%),radial-gradient(500px 600px at 50% 100%,color-mix(in srgb,var(--accent) 8%,transparent),transparent 60%);}`;
    70→    case 'grid': return `.ambient{opacity:.45;background-image:linear-gradient(var(--border) 1px,transparent 1px),linear-gradient(90deg,var(--border) 1px,transparent 1px);background-size:56px 56px;}`;
    71→    case 'rings': return `.ambient{opacity:.5;background:repeating-radial-gradient(circle at 85% 10%,transparent 0 46px,var(--border) 46px 47px);}`;
    72→    case 'waves': return `.ambient{opacity:.5;background:repeating-linear-gradient(115deg,transparent 0 38px,var(--border) 38px 39px);}`;
    73→    case 'dots': return `.ambient{opacity:.55;background-image:radial-gradient(color-mix(in srgb,var(--inkSoft) 33%,transparent) 1.2px,transparent 1.2px);background-size:26px 26px;}`;
    74→    default: return `.ambient{opacity:.5;background:radial-gradient(120% 90% at 50% -20%,transparent 60%,color-mix(in srgb,var(--accent) 9%,transparent) 100%);}`;
