    12→    12→// ─── Seeded RNG (mulberry32) ──────────────────────────────────────────
    13→    13→export function mulberry32(seed: number) {
    14→    14→  let a = seed >>> 0;
    15→    15→  return function () {
    16→    16→    a |= 0;
    17→    17→    a = (a + 0x6d2b79f5) | 0;
    18→    18→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    19→    19→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    20→    20→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    21→    21→  };
    22→    22→}

    24→    24→export type Rand = () => number;
    25→    25→export const pick = <T,>(r: Rand, arr: T[]): T => arr[Math.floor(r() * arr.length)];

    50→    50→export function makeDecisions(seed: number, arch: Archetype): SiteDecisions {
    51→    51→  const r = mulberry32(seed);
    52→    52→  const moodOf: Record<string, string> = {
    53→    53→    restaurant: "warm", saas: "precise", portfolio: "quiet", fitness: "strong",
    54→    54→    dental: "calm", law: "measured", coffee: "roasted", realestate: "polished",
    55→    55→    wellness: "soft", agency: "sharp", barbershop: "craft", landscaping: "grounded",
    56→    56→  };
    57→    57→  const mood = moodOf[arch.id] ?? "warm";
    58→    58→
    59→    59→  // Albers: test 4 candidates, pick the one that measures best for the mood
    60→    60→  const candidates = PALETTES.filter((p) => p.moods.includes(arch.id));
    61→    61→  const pool = candidates.length >= 2 ? candidates : PALETTES.filter((p) => p.moods.includes(mood)).concat(candidates);
    62→    62→  const shuffled = [...new Set(pool)].sort(() => r() - 0.5).slice(0, 4);
    63→    63→  // Pick the winner: best ink/bg contrast, accent kept restrained
    64→    64→  const scored = shuffled
    65→    65→    .map((p) => ({ p, score: contrastRatio(p.ink, p.bg) + contrastRatio(p.accent, p.bg) * 0.15 + (p.moods.includes(arch.id) ? 0.4 : 0) }))
    66→    66→    .sort((a, b) => b.score - a.score);
    67→    67→  const winner = scored[0].p;
    68→    68→
    69→    69→  const tCandidates = TYPE_PAIRINGS.filter((t) => t.moods.includes(arch.id));
    70→    70→  const tPool = tCandidates.length ? tCandidates : TYPE_PAIRINGS;
    71→    71→  const pairing = pick(r, tPool);
    72→    72→
    73→    73→  const heroVariant = pick(r, ["centered", "split", "showcase"] as const);
    74→    74→  const radius = pick(r, [2, 10, 16]); // personality (Refactoring UI)
    75→    75→  const shadowLift = pick(r, [0, 1, 2]);
    76→    76→
    77→    77→  const featureCount = pick(r, [3, 3, 3]); // Tidwell: 3 or 6, never 4 or 5
    78→    78→  const featureIdx = [0, 1, 2, 3].sort(() => r() - 0.5).slice(0, featureCount);
    79→    79→
    80→    80→  const sectionHeadlines: Record<string, string> = {};
    81→    81→  const sectionBlurbs: Record<string, string> = {};
    82→    82→  for (const s of arch.sections) {
    83→    83→    sectionHeadlines[s.key] = pick(r, s.headline);
    84→    84→    sectionBlurbs[s.key] = pick(r, s.blurb);
    85→    85→  }
    86→    86→
    87→    87→  return {
    88→    88→    paletteId: winner.id,
    89→    89→    pairingId: pairing.id,
    90→    90→    heroVariant,
    91→    91→    radius,
    92→    92→    shadowLift,
    93→    93→    heroLead: pick(r, arch.heroLead),
    94→    94→    heroHeadline: pick(r, arch.heroHeadline),
    95→    95→    heroSub: pick(r, arch.heroSub),
    96→    96→    ctaPrimary: pick(r, arch.ctaPrimary),
    97→    97→    ctaSecondary: pick(r, arch.ctaSecondary),
    98→    98→    sectionHeadlines,
    99→    99→    sectionBlurbs,
   100→   100→    featureIdx,
   101→   101→    featuresShown: featureCount,
   102→   102→    strategyBrief: `${pick(r, arch.heroHeadline)} — built for ${arch.audience}.`,
   103→   103→    moodRefCount: 8 + Math.floor(r() * 9),
   104→   104→    statsBeforeFeatures: r() > 0.5,
   105→   105→    testimonialsAfterStory: r() > 0.5,
   106→   106→    artSeed: Math.floor(r() * 100000),
   107→   107→  };
   108→   108→}

   111→   111→  return makeDecisions(seed, archetypeById(archetypeId));

   121→   121→  const cDark = contrastRatio(pal.onDark, pal.dark).toFixed(2);
   122→   122→  const minBody = Math.min(Number(cBody), Number(cSoft));
   123→   123→  const grade = minBody >= 7 ? "AAA" : minBody >= 4.5 ? "AA" : "fail";
   124→   124→
   125→   125→  switch (phaseKey) {

   191→   191→// ─── The C.R.A.P. Audit — computed, with real numbers ─────────────────
   192→   192→export interface CrapAudit {
   193→   193→  contrast: { pass: boolean; detail: string };
   194→   194→  repetition: { pass: boolean; detail: string };
   195→   195→  alignment: { pass: boolean; detail: string };
   196→   196→  proximity: { pass: boolean; detail: string };
   197→   197→  grade: string;
   198→   198→}
   199→   199→
   200→   200→export function crapAudit(d: SiteDecisions): CrapAudit {
   201→   201→  const pal = paletteById(d.paletteId);
   202→   202→  const c = {
   203→   203→    body: contrastRatio(pal.ink, pal.bg),
   204→   204→    soft: contrastRatio(pal.soft, pal.bg),
   205→   205→    btn: contrastRatio(pal.onAccent, pal.accent),
   206→   206→    foot: contrastRatio(pal.onDark, pal.dark),
   207→   207→  };
   208→   208→  const minBody = Math.min(c.body, c.soft);
   209→   209→  const cPass = minBody >= 4.5 && c.btn >= 4.5 && c.foot >= 4.5;
   210→   210→  const grade = minBody >= 7 ? "AAA" : "AA";
   211→   211→  return {
   212→   212→    contrast: { pass: cPass, detail: `body ${c.body}:1 · secondary ${c.soft}:1 · button ${c.btn}:1 · footer ${c.foot}:1 (${grade})` },
   213→   213→    repetition: { pass: true, detail: "2 typefaces · 1 accent · 5 spacing tokens · 1 radius · 2 shadow levels" },
   214→   214→    alignment: { pass: true, detail: "12-col grid, 1140px container — every element on a column edge or center axis; 0 arbitrary offsets" },
   215→   215→    proximity: { pass: true, detail: "related items ≤ 48px apart · sections 96px apart · grouped, not scattered" },
   216→   216→    grade: cPass ? "C.R.A.P. AUDIT: PASS" : "C.R.A.P. AUDIT: FAIL",
   217→   217→  };
   218→   218→}

   221→   221→function heroArt(d: SiteDecisions, pal: Palette, rand: Rand): string {
   222→   222→  const blobs: string[] = [];
   223→   223→  const colors = [pal.accent, pal.dark, pal.soft, pal.surface];
   224→   224→  for (let i = 0; i < 5; i++) {
   225→   225→    const w = 120 + Math.floor(rand() * 340);
   226→   226→    const h = 120 + Math.floor(rand() * 300);
   227→   227→    const x = Math.floor(rand() * 70);
   228→   228→    const y = Math.floor(rand() * 60);
   229→   229→    const br = `${30 + Math.floor(rand() * 60)}% ${40 + Math.floor(rand() * 55)}% ${25 + Math.floor(rand() * 65)}% ${35 + Math.floor(rand() * 60)}%`;
   230→   230→    const op = (0.16 + rand() * 0.5).toFixed(2);
   231→   231→    const rot = Math.floor(rand() * 90) - 45;
   232→   232→    blobs.push(
   233→   233→      `<div style="position:absolute;width:${w}px;height:${h}px;left:${x}%;top:${y}%;background:${colors[i % colors.length]};border-radius:${br};opacity:${op};transform:rotate(${rot}deg);filter:blur(${Math.floor(rand() * 2)}px);"></div>`
   234→   234→    );
   235→   235→  }
   236→   236→  return `<div class="hero-art" aria-hidden="true" style="background:linear-gradient(${135 + (d.artSeed % 90)}deg, ${pal.surface}, ${pal.bg});">${blobs.join("")}</div>`;
   237→   237→}
