    67→export default function ControlPage() {
    68→  const [state, setState] = useState<State | null>(null);
    69→  const [busy, setBusy] = useState<string>('');
    70→  const [ownerDraft, setOwnerDraft] = useState('');

    72→  const [values, setValues] = useState<Record<string, unknown> | null>(null);
    73→  const { toast } = useToast();
    74→
    75→  const refresh = useCallback(async () => {
    76→    try {
    77→      const res = await fetch('/api/beast', { cache: 'no-store' });
    78→      const json = (await res.json()) as State;
    79→      setState(json);
    80→      setOwnerDraft((d) => d || json.owner);
    81→    } catch { /* the studio keeps running even if a poll fails */ }
    82→  }, []);
    83→
    84→  useEffect(() => { refresh(); }, [refresh]);
    85→  useEffect(() => {
    86→    const t = setInterval(refresh, 15000);
    87→    return () => clearInterval(t);
    88→  }, [refresh]);
    89→
    90→  const act = useCallback(async (body: Record<string, unknown>, label: string) => {
    91→    setBusy(label);
    92→    try {
    93→      const res = await fetch('/api/beast', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
    94→      const json = await res.json();
    95→      if (json.error) throw new Error(json.error);
    96→      setState(json as State);
    97→      toast({ title: label, description: Array.isArray(json.jobs) ? json.jobs.slice(0, 2).join(' · ') : 'Done.' });
    98→    } catch (e) {
    99→      toast({ title: `${label} failed`, description: String((e as Error).message), variant: 'destructive' });
   100→    } finally { setBusy(''); }
   101→  }, [toast]);
   102→

   104→    setEditorSite(siteId);
   105→    setValues(null);
   106→    try {
   107→      const res = await fetch('/api/beast', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'getValues', siteId }) });
   108→      const json = await res.json();
   109→      setValues(json.values ?? {});
   110→    } catch { toast({ title: 'Could not load values', variant: 'destructive' }); }
   111→  };
   112→
   113→  const saveValues = async () => {
   114→    if (!editorSite || !values) return;
   115→    setBusy('Save values');
   116→    try {
   117→      const res = await fetch('/api/beast', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'setValues', siteId: editorSite, values }) });
   118→      const json = await res.json();
   119→      if (!res.ok || json.ok === false) throw new Error((json.issues ?? ['invalid values']).join('; '));
   120→      setState(json as State);
   121→      toast({ title: 'Values saved', description: 'Every built page re-rendered with the new facts.' });
   122→    } catch (e) {
   123→      toast({ title: 'Save rejected', description: String((e as Error).message), variant: 'destructive' });
   124→    } finally { setBusy(''); }
   125→  };
   126→
   127→  if (!state) {
   128→    return (
   129→      <div className="min-h-screen bg-zinc-950 text-zinc-100 flex items-center justify-center">
   130→        <div className="flex items-center gap-3 text-zinc-400"><RefreshCw className="h-5 w-5 animate-spin" /> Waking the Beast…</div>
   131→      </div>
   132→    );
   133→  }
   134→
   135→  const inStudio = state.sites.filter((s) => s.status === 'in-studio');
   136→  const forSale = state.sites.filter((s) => s.status === 'for-sale');
   137→  const allTasks = state.sites.flatMap((s) => s.tasks);

   139→  const verdicts = state.transcript.reduce<Record<string, number>>((m, t) => { m[t.verdict] = (m[t.verdict] ?? 0) + 1; return m; }, {});
   140→  const aiTurns = state.transcript.filter((t) => t.mode === 'ai').length;

   157→    if (platform === 'youtube') return `https://youtube.com/@${h}`;

   168→            <Gauge className="h-5 w-5 text-amber-400" />
   169→            <h1 className="font-semibold tracking-tight">Mission Control</h1>
   170→            <Badge variant="outline" className="text-zinc-400">v{state.version}</Badge>
   171→          </div>
   172→          <div className="ml-auto flex items-center gap-2 text-xs text-zinc-400 flex-wrap">
   173→            <Badge variant="outline" className="gap-1"><Clock className="h-3 w-3" /> DAY {state.day}</Badge>
   174→            <Badge variant="outline" className="gap-1"><Database className="h-3 w-3" /> GitHub {state.github.configured ? 'connected' : 'not connected'}</Badge>
   175→            <Badge variant="outline" className="gap-1"><Sparkles className="h-3 w-3" /> AI {state.ai.enabled ? 'enabled' : 'solver floor'}</Badge>
   176→            <Badge variant="outline" className="gap-1"><Braces className="h-3 w-3" /> {state.layoutCount} layouts</Badge>

   182→      <main className="mx-auto max-w-7xl px-4 py-6 space-y-6">

   186→            { label: 'Overall progress', value: `${avgProgress}%` },
   187→            { label: 'In studio', value: inStudio.length },
   188→            { label: 'On the shelf', value: forSale.length },
   189→            { label: 'Micro-tasks done', value: accepted.length },
   190→            { label: 'Avg precision', value: avgPrecision ? `${Math.round(avgPrecision * 100)}%` : '—' },
   191→            { label: 'QA contrast', value: qaPass === null ? '—' : `${qaPass}%` },
   192→          ].map((s) => (
   193→            <Card key={s.label} className="bg-zinc-900 border-zinc-800">
   194→              <CardContent className="p-4">
   195→                <div className="text-xs text-zinc-500">{s.label}</div>

   197→              </CardContent>
   198→            </Card>

   202→        <Tabs defaultValue="progress">

   218→            <div className="grid md:grid-cols-2 gap-4">
   219→              {state.sites.map((s) => (
